from sklearn.tree import DecisionTreeRegressor

X = [[10000, 2, 1000],
     [20000, 3, 1200],
     [30000, 4, 1500],
     [40000, 5, 1800],
     [50000, 6, 2000]]

y = [800000, 700000, 600000, 500000, 400000]

model = DecisionTreeRegressor(random_state=0)

model.fit(X, y)

a = float(input("Enter mileage: "))
b = float(input("Enter car age: "))
c = float(input("Enter engine size: "))

car = [[a, b, c]]

result = model.predict(car)

print("Predicted Car Price:", result[0])

path = model.decision_path(car)

print("Decision Path:")

for node in path.indices:
    if model.tree_.children_left[node] != model.tree_.children_right[node]:
        feature = model.tree_.feature[node]
        value = model.tree_.threshold[node]

        if car[0][feature] <= value:
            print("Feature", feature, "<=", value)
        else:
            print("Feature", feature, ">", value)
