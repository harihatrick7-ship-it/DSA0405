from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error

X = [[1000, 80], [1200, 90], [1500, 100],
     [1800, 120], [2000, 150]]

y = [500000, 600000, 750000, 900000, 1000000]

model = LinearRegression()

model.fit(X, y)

a = float(input("Enter engine size: "))
b = float(input("Enter horsepower: "))

result = model.predict([[a, b]])

print("Predicted Car Price:", result[0])

pred = model.predict(X)

print("Mean Squared Error:", mean_squared_error(y, pred))

print("Feature Importance:")
print(model.coef_)
