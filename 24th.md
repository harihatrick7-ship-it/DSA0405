from sklearn.neighbors import KNeighborsClassifier

X = [[1,2], [2,3], [3,1], [6,7], [7,8], [8,6]]
y = [0, 0, 0, 1, 1, 1]

k = int(input("Enter K value: "))

model = KNeighborsClassifier(n_neighbors=k)

model.fit(X, y)

a = float(input("Enter first symptom: "))
b = float(input("Enter second symptom: "))

result = model.predict([[a,b]])

print("Prediction:", result[0])

if result[0] == 1:
    print("Patient has condition")
else:
    print("Patient has no condition")
