import matplotlib.pyplot as plt

month = [1, 2, 3, 4, 5, 6]
temp = [25, 28, 30, 32, 29, 27]

plt.plot(month, temp)
plt.xlabel("Month")
plt.ylabel("Temperature")
plt.title("Monthly Temperature")
plt.show()
