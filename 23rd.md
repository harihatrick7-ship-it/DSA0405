import matplotlib.pyplot as plt
from scipy import stats

drug = [10,12,14,15,13,11,16,14]
placebo = [5,6,7,8,6,5,9,7]

t, p = stats.ttest_ind(drug, placebo)

print("T-value:", t)
print("P-value:", p)

if p < 0.05:
    print("Treatment is effective")
else:
    print("Treatment is not effective")

plt.bar(['Drug', 'Placebo'],
        [sum(drug)/len(drug), sum(placebo)/len(placebo)])

plt.ylabel("Mean Reduction")
plt.title("Drug vs Placebo")
plt.show()
