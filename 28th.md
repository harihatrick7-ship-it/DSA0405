from sklearn.cluster import KMeans

X = [[100, 2], [150, 3], [200, 2],
     [800, 8], [850, 9], [900, 8]]

model = KMeans(n_clusters=2, random_state=0, n_init=10)

model.fit(X)

a = float(input("Enter amount spent: "))
b = float(input("Enter number of purchases: "))

result = model.predict([[a,b]])

print("Customer belongs to Group:", result[0])
