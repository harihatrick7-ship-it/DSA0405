import matplotlib.pyplot as plt

month = [1, 2, 3, 4, 5]
sales = [100, 200, 150, 300, 250]

plt.plot(month, sales)
plt.xlabel("Month")
plt.ylabel("Sales")
plt.title("Monthly Sales")
plt.show()
