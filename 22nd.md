import pandas as pd
from scipy import stats
import numpy as np

df = pd.read_csv("customer_reviews.csv")

data = df['rating']

mean = data.mean()
sd = data.std()
n = len(data)

t = stats.t.ppf(0.975, n-1)

error = t * sd / np.sqrt(n)

print("Mean Rating:", mean)
print("Lower Limit:", mean-error)
print("Upper Limit:", mean+error)
