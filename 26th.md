from sklearn.linear_model import LinearRegression

X = [[1000, 2], [1500, 3], [2000, 3],
     [2500, 4], [3000, 4]]

y = [200000, 300000, 400000, 500000, 600000]

model = LinearRegression()

model.fit(X, y)

area = float(input("Enter house area: "))
bedrooms = int(input("Enter number of bedrooms: "))

result = model.predict([[area, bedrooms]])

print("Predicted House Price:", result[0])
