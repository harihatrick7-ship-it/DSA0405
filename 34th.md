from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

X = [[20, 110], [25, 120], [30, 115],
     [50, 150], [55, 160], [60, 170]]

y = ['Good', 'Good', 'Good', 'Bad', 'Bad', 'Bad']

model = KNeighborsClassifier(n_neighbors=3)

model.fit(X, y)

pred = model.predict(X)

print("Predictions:", pred)

print("Accuracy:", accuracy_score(y, pred))
print("Precision:", precision_score(y, pred, pos_label='Good'))
print("Recall:", recall_score(y, pred, pos_label='Good'))
print("F1 Score:", f1_score(y, pred, pos_label='Good'))

a = float(input("Enter age: "))
b = float(input("Enter blood pressure: "))

result = model.predict([[a, b]])

print("Treatment Outcome:", result[0])
