import numpy as np
import pandas as pd
from scipy import stats

df = pd.read_csv("rare_elements.csv")

data = df.iloc[:, 0].dropna().values

n = int(input("Enter sample size: "))
level = float(input("Enter confidence level: "))
precision = float(input("Enter precision: "))

sample = data[:n]

mean = np.mean(sample)
sd = np.std(sample, ddof=1)

t = stats.t.ppf((1 + level/100)/2, n-1)

error = t * sd / np.sqrt(n)

print("Mean:", mean)
print("Lower Limit:", mean-error)
print("Upper Limit:", mean+error)
print("Margin of Error:", error)

if error <= precision:
    print("Precision achieved")
else:
    print("Precision not achieved")
