import pandas as pd

data = {
    'City': ['Chennai', 'Chennai', 'Chennai',
             'Delhi', 'Delhi', 'Delhi',
             'Mumbai', 'Mumbai', 'Mumbai'],

    'Temp': [30, 32, 34, 20, 25, 30, 28, 29, 30]
}

df = pd.DataFrame(data)

print("Mean Temperature:")
print(df.groupby('City')['Temp'].mean())

print("Standard Deviation:")
print(df.groupby('City')['Temp'].std())

print("Temperature Range:")
r = df.groupby('City')['Temp'].max() - df.groupby('City')['Temp'].min()
print(r)

print("Highest Range:", r.idxmax())
