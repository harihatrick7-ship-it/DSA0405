from sklearn.linear_model import LogisticRegression

X = [[10, 1], [20, 2], [30, 3],
     [70, 6], [80, 7], [90, 8]]

y = [0, 0, 0, 1, 1, 1]

model = LogisticRegression()

model.fit(X, y)

a = float(input("Enter usage minutes: "))
b = float(input("Enter contract duration: "))

result = model.predict([[a,b]])

if result[0] == 1:
    print("Customer will leave")
else:
    print("Customer will stay")
